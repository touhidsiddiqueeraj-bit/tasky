rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ── Users ────────────────────────────────────────────────────────────────
    match /users/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // ── Groups ───────────────────────────────────────────────────────────────
    match /groups/{code} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.resource.data.supervisorUid == request.auth.uid;
      allow update: if request.auth != null && (
        resource.data.supervisorUid == request.auth.uid
        ||
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members'])
         && request.auth.uid == request.resource.data.members[request.resource.data.members.size()-1].uid)
      );
      allow delete: if request.auth != null
        && resource.data.supervisorUid == request.auth.uid;

      // ── Per-member task snapshots ──────────────────────────────────────────
      match /tasks/{memberId} {
        allow write: if request.auth != null && (
          request.auth.uid == memberId
          || get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
        );
        allow read: if request.auth != null && (
          get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
          || request.auth.uid == memberId
        );
      }

      // ── Message board ──────────────────────────────────────────────────────
      match /messages/{messageId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null
          && request.resource.data.authorUid == request.auth.uid;
        allow update: if request.auth != null
          && resource.data.authorUid == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.authorUid == request.auth.uid
          || get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
        );
        match /replies/{replyId} {
          allow read: if request.auth != null;
          allow create: if request.auth != null
            && request.resource.data.authorUid == request.auth.uid;
          allow update: if request.auth != null
            && resource.data.authorUid == request.auth.uid;
          allow delete: if request.auth != null && (
            resource.data.authorUid == request.auth.uid
            || get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
          );
        }
      }

      // ── Whiteboard ─────────────────────────────────────────────────────────
      match /whiteboard/{docId} {
        allow read, create, update: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(code));
        allow delete: if false;
      }

      // ── Visitor Comments (read-only board) ─────────────────────────────────
      // Reads/writes require (anonymous) auth so drive-by scrapers and bots
      // can't enumerate or spam boards. The app signs visitors in anonymously.
      // hasOnly (not just hasAll) blocks smuggled fields like taskText/authorUid;
      // taskNumber is range-checked so it can't be used as an ID oracle.
      match /visitor-comments/{commentId} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(code));
        allow create: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(code))
          && request.resource.data.keys().hasOnly(['text', 'authorName', 'taskNumber', 'taskText', 'authorEmail', 'createdAt'])
          && request.resource.data.text is string
          && request.resource.data.text.size() >= 1
          && request.resource.data.text.size() <= 2000
          && request.resource.data.authorName is string
          && request.resource.data.authorName.size() >= 1
          && request.resource.data.authorName.size() <= 100
          && request.resource.data.taskNumber is number
          && request.resource.data.taskNumber >= 0
          && request.resource.data.taskNumber <= 1000000
          && (!('taskText' in request.resource.data) || (request.resource.data.taskText is string && request.resource.data.taskText.size() <= 500))
          && (!('authorEmail' in request.resource.data) || (request.resource.data.authorEmail is string && request.resource.data.authorEmail.size() <= 320));
        allow update, delete: if false;
        match /replies/{replyId} {
          allow read: if request.auth != null
            && exists(/databases/$(database)/documents/groups/$(code));
          allow create: if request.auth != null
            && exists(/databases/$(database)/documents/groups/$(code))
            && request.resource.data.keys().hasOnly(['text', 'authorName', 'createdAt'])
            && request.resource.data.text is string
            && request.resource.data.text.size() >= 1
            && request.resource.data.text.size() <= 2000
            && request.resource.data.authorName is string
            && request.resource.data.authorName.size() >= 1
            && request.resource.data.authorName.size() <= 100;
          allow update: if false;
          allow delete: if request.auth != null
            && get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid;
        }
      }
    }

    // ── Notifications ─────────────────────────────────────────────────────────
    // hasOnly blocks smuggled fields; fromHandle is pinned to the sender's own
    // users-doc handle when that doc exists (users can only write their own
    // doc), so attackers can't forge toasts from other people. No-doc senders
    // (handle never set) are still allowed — handle is required client-side
    // before any notification is sent.
    match /notifications/{notifId} {
      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['toUid', 'fromHandle', 'groupCode', 'taskText', 'taskId', 'commentText', 'priority', 'dueDate', 'type', 'createdAt', 'read'])
        && request.resource.data.toUid is string
        && request.resource.data.fromHandle is string
        && request.resource.data.fromHandle.size() >= 1
        && request.resource.data.fromHandle.size() <= 100
        && request.resource.data.read == false
        && (!('taskText' in request.resource.data) || (request.resource.data.taskText is string && request.resource.data.taskText.size() <= 500))
        && (!('commentText' in request.resource.data) || (request.resource.data.commentText is string && request.resource.data.commentText.size() <= 2000))
        && (!exists(/databases/$(database)/documents/users/$(request.auth.uid))
            || !('handle' in get(/databases/$(database)/documents/users/$(request.auth.uid)).data)
            || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.handle == request.resource.data.fromHandle);
      allow read, update: if request.auth != null
        && resource.data.toUid == request.auth.uid;
      allow delete: if false;
    }

    // ── Voice Sessions ────────────────────────────────────────────────────────
    match /voice_sessions/{groupCode} {
      allow read: if request.auth != null;
      allow write: if false;

      // ── Presence ──────────────────────────────────────────────────────────
      match /participants/{uid} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode));
        allow create, update: if request.auth != null
          && request.auth.uid == uid
          && request.resource.data.uid == request.auth.uid;
        allow delete: if request.auth != null && (
          request.auth.uid == uid
          || get(/databases/$(database)/documents/groups/$(groupCode)).data.supervisorUid == request.auth.uid
        );
      }

      // ── WebRTC Signals ────────────────────────────────────────────────────
      match /signals/{signalId} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode));
        allow create: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode))
          && request.resource.data.from == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.from == request.auth.uid
          || resource.data.to == request.auth.uid
        );
        allow update: if false;
      }

      // ── Ring Notifications ────────────────────────────────────────────────
      match /ring/{ringId} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode));
        allow create, update: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode))
          && request.resource.data.callerUid == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.callerUid == request.auth.uid
          || get(/databases/$(database)/documents/groups/$(groupCode)).data.supervisorUid == request.auth.uid
        );
      }
    }
  }
}
