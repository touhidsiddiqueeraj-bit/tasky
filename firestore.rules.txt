rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ── Users ────────────────────────────────────────────────────────────────
    match /users/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // ── Groups ───────────────────────────────────────────────────────────────
    match /groups/{code} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.resource.data.supervisorUid == request.auth.uid;
      allow update: if request.auth != null && (
        resource.data.supervisorUid == request.auth.uid
        ||
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members'])
         && request.auth.uid == request.resource.data.members[request.resource.data.members.size()-1].uid)
      );
      allow delete: if request.auth != null
        && resource.data.supervisorUid == request.auth.uid;

      // ── Per-member task snapshots ──────────────────────────────────────────
      match /tasks/{memberId} {
        allow write: if request.auth != null && (
          request.auth.uid == memberId
          || get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
        );
        allow read: if request.auth != null && (
          get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
          || request.auth.uid == memberId
        );
      }

      // ── Message board ──────────────────────────────────────────────────────
      match /messages/{messageId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null
          && request.resource.data.authorUid == request.auth.uid;
        allow update: if request.auth != null
          && resource.data.authorUid == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.authorUid == request.auth.uid
          || get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
        );
        match /replies/{replyId} {
          allow read: if request.auth != null;
          allow create: if request.auth != null
            && request.resource.data.authorUid == request.auth.uid;
          allow update: if request.auth != null
            && resource.data.authorUid == request.auth.uid;
          allow delete: if request.auth != null && (
            resource.data.authorUid == request.auth.uid
            || get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid
          );
        }
      }

      // ── Whiteboard ─────────────────────────────────────────────────────────
      match /whiteboard/{docId} {
        allow read, create, update: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(code));
        allow delete: if false;
      }

      // ── Visitor Comments (read-only board) ─────────────────────────────────
      // Public read/create allowed so clients can leave feedback without signing in.
      // Rate-limit: each comment doc includes a timestamp and the group doc exists check.
      match /visitor-comments/{commentId} {
        allow read: if true;
        allow create: if true
          && request.resource.data.keys().hasAll(['text', 'authorName', 'taskNumber', 'createdAt'])
          && request.resource.data.text is string
          && request.resource.data.text.size() <= 2000
          && request.resource.data.authorName is string
          && request.resource.data.authorName.size() <= 100
          && request.resource.data.taskNumber is number;
        match /replies/{replyId} {
          allow read: if true;
          allow create: if true
            && request.resource.data.keys().hasAll(['text', 'authorName', 'createdAt'])
            && request.resource.data.text is string
            && request.resource.data.text.size() <= 2000
            && request.resource.data.authorName is string
            && request.resource.data.authorName.size() <= 100;
          allow delete: if request.auth != null
            && get(/databases/$(database)/documents/groups/$(code)).data.supervisorUid == request.auth.uid;
        }
      }
    }

    // ── Notifications ─────────────────────────────────────────────────────────
    match /notifications/{notifId} {
      allow create: if request.auth != null;
      allow read, update: if request.auth != null
        && resource.data.toUid == request.auth.uid;
      allow delete: if false;
    }

    // ── Voice Sessions ────────────────────────────────────────────────────────
    match /voice_sessions/{groupCode} {
      allow read: if request.auth != null;
      allow write: if false;

      // ── Presence ──────────────────────────────────────────────────────────
      match /participants/{uid} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode));
        allow create, update: if request.auth != null
          && request.auth.uid == uid
          && request.resource.data.uid == request.auth.uid;
        allow delete: if request.auth != null && (
          request.auth.uid == uid
          || get(/databases/$(database)/documents/groups/$(groupCode)).data.supervisorUid == request.auth.uid
        );
      }

      // ── WebRTC Signals ────────────────────────────────────────────────────
      match /signals/{signalId} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode));
        allow create: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode))
          && request.resource.data.from == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.from == request.auth.uid
          || resource.data.to == request.auth.uid
        );
        allow update: if false;
      }

      // ── Ring Notifications ────────────────────────────────────────────────
      match /ring/{ringId} {
        allow read: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode));
        allow create, update: if request.auth != null
          && exists(/databases/$(database)/documents/groups/$(groupCode))
          && request.resource.data.callerUid == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.callerUid == request.auth.uid
          || get(/databases/$(database)/documents/groups/$(groupCode)).data.supervisorUid == request.auth.uid
        );
      }
    }
  }
}
